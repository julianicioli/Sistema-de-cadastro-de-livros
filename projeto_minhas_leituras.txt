# Minhas Leituras — gerenciador de livros e mangás

App mobile em **React Native (Expo)** integrado ao **Supabase** (PostgreSQL). Permite cadastrar obras
(livros ou mangás) com quantidade de volumes/capítulos e total de páginas, marcar o progresso de leitura
e calcular automaticamente o percentual concluído.

## Funcionalidades
- Cadastrar, editar e excluir obras (título, autor, tipo, volumes, capítulos, páginas)
- Marcar progresso: páginas lidas, capítulo atual e volume atual (botões +/− ou digitando)
- Percentual concluído calculado automaticamente:
  - no app, em tempo real (`src/utils/progresso.js`)
  - no banco, como coluna gerada `percentual` (`supabase/schema.sql`)
- Botão "Marcar como concluído"
- Lista com filtro (Todos / Livros / Mangás), status (Não iniciado / Lendo / Concluído) e média geral

## Como rodar
1. Crie um projeto em https://supabase.com.
2. No Supabase, abra **SQL Editor**, cole o conteúdo de `supabase/schema.sql` e clique em **Run**.
3. Copie `.env.example` para `.env` e preencha a URL do projeto e a chave publishable/anon
   (Project Settings → Data API e API Keys, ou o botão **Connect**).
   A URL deve terminar em `.supabase.co`. Se vier com `/rest/v1/`, o app remove automaticamente.
4. Instale e inicie:
   ```bash
   npm install
   npx expo install --fix   # alinha as versões das libs nativas com o SDK do Expo
   npx expo start
   ```
5. Abra no celular com o app **Expo Go** (lendo o QR code) ou em um emulador Android (`a`).

## Estrutura
```
App.js                         navegação (stack)
src/lib/supabase.js            cliente Supabase
src/services/obras.js          CRUD da tabela "obras"
src/utils/progresso.js         cálculo do percentual e status
src/components/BarraProgresso.js
src/screens/ListaObrasScreen.js   lista + resumo + filtros
src/screens/DetalheObraScreen.js  marcar progresso / concluir / excluir
src/screens/FormObraScreen.js     cadastro e edição
supabase/schema.sql            tabela, coluna gerada, trigger e RLS
```

## Regra do percentual
`percentual = min(100, paginas_lidas / total_paginas × 100)`, arredondado a 1 casa decimal.
Para obras com capítulos, a tela de detalhe também mostra o percentual por capítulos.

-- Execute este script no Supabase: SQL Editor > New query > Run

create table if not exists public.obras (
  id               bigint generated always as identity primary key,
  titulo           text    not null,
  autor            text,
  tipo             text    not null default 'livro' check (tipo in ('livro', 'manga')),
  total_volumes    integer not null default 0 check (total_volumes >= 0),
  total_capitulos  integer not null default 0 check (total_capitulos >= 0),
  total_paginas    integer not null check (total_paginas > 0),
  volume_atual     integer not null default 0 check (volume_atual >= 0),
  capitulo_atual   integer not null default 0 check (capitulo_atual >= 0),
  paginas_lidas    integer not null default 0 check (paginas_lidas >= 0),
  -- Percentual calculado automaticamente pelo banco a partir das páginas lidas
  percentual       numeric(5,1) generated always as (
                     least(100, round(paginas_lidas * 100.0 / total_paginas, 1))
                   ) stored,
  criado_em        timestamptz not null default now(),
  atualizado_em    timestamptz not null default now(),
  constraint paginas_validas check (paginas_lidas <= total_paginas)
);

-- Atualiza "atualizado_em" a cada alteração
create or replace function public.set_atualizado_em()
returns trigger language plpgsql as $$
begin
  new.atualizado_em := now();
  return new;
end;
$$;

drop trigger if exists trg_obras_atualizado on public.obras;
create trigger trg_obras_atualizado
before update on public.obras
for each row execute function public.set_atualizado_em();

-- Row Level Security: liberado para a chave anon (projeto acadêmico, sem login).
alter table public.obras enable row level security;

drop policy if exists "acesso publico" on public.obras;
create policy "acesso publico" on public.obras
  for all to anon, authenticated
  using (true) with check (true);

import { View, Text, StyleSheet } from 'react-native';
import { cores } from '../theme';

export default function BarraProgresso({ percentual, altura = 10, mostrarTexto = true }) {
  const p = Math.max(0, Math.min(100, Number(percentual) || 0));
  const concluido = p >= 100;
  return (
    <View>
      <View style={[styles.trilho, { height: altura, borderRadius: altura / 2 }]}>
        <View
          style={{
            width: `${p}%`,
            height: '100%',
            backgroundColor: concluido ? cores.estrela : cores.primaria,
            borderRadius: altura / 2,
          }}
        />
      </View>
      {mostrarTexto && (
        <Text style={[styles.texto, concluido && { color: cores.estrela }]}>
          {concluido ? '★ ' : '✦ '}
          {p.toFixed(1)}% concluído
        </Text>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  trilho: { backgroundColor: cores.trilho, overflow: 'hidden' },
  texto: { marginTop: 4, fontSize: 12, color: cores.textoSecundario },
});

import { useEffect, useRef } from 'react';
import { View, Animated, StyleSheet } from 'react-native';
import { cores } from '../theme';

const SIMBOLOS = ['✦', '✧', '⋆', '★', '·'];

// Posições "aleatórias" fixas (geradas uma vez), para as estrelas não mudarem de lugar
function gerarEstrelas(qtd) {
  let seed = 7;
  const rand = () => ((seed = (seed * 9301 + 49297) % 233280) / 233280);
  return Array.from({ length: qtd }, (_, i) => ({
    id: i,
    simbolo: SIMBOLOS[Math.floor(rand() * SIMBOLOS.length)],
    top: `${rand() * 100}%`,
    left: `${rand() * 100}%`,
    tamanho: 8 + rand() * 14,
    dourada: rand() < 0.25,
    duracao: 1500 + rand() * 2500,
    atraso: rand() * 2000,
  }));
}

const ESTRELAS = gerarEstrelas(36);

function Estrela({ e }) {
  const brilho = useRef(new Animated.Value(0.25)).current;

  useEffect(() => {
    const anim = Animated.loop(
      Animated.sequence([
        Animated.delay(e.atraso),
        Animated.timing(brilho, { toValue: 0.9, duration: e.duracao, useNativeDriver: true }),
        Animated.timing(brilho, { toValue: 0.25, duration: e.duracao, useNativeDriver: true }),
      ])
    );
    anim.start();
    return () => anim.stop();
  }, []);

  return (
    <Animated.Text
      style={{
        position: 'absolute',
        top: e.top,
        left: e.left,
        fontSize: e.tamanho,
        color: e.dourada ? cores.estrela : cores.destaque,
        opacity: brilho,
      }}
    >
      {e.simbolo}
    </Animated.Text>
  );
}

// Envolve a tela com o céu roxo estrelado
export default function Fundo({ children }) {
  return (
    <View style={styles.fundo}>
      <View style={StyleSheet.absoluteFill} pointerEvents="none">
        {ESTRELAS.map((e) => <Estrela key={e.id} e={e} />)}
      </View>
      {children}
    </View>
  );
}

const styles = StyleSheet.create({
  fundo: { flex: 1, backgroundColor: cores.fundo, overflow: 'hidden' },
});

import { useCallback, useState } from 'react';
import { ScrollView, View, Text, TextInput, TouchableOpacity, StyleSheet, ActivityIndicator } from 'react-native';
import { avisar, confirmar } from '../utils/alerta';
import { useFocusEffect } from '@react-navigation/native';
import { buscarObra, atualizarObra, excluirObra } from '../services/obras';
import BarraProgresso from '../components/BarraProgresso';
import Fundo from '../components/Fundo';
import { calcularPercentual, percentualCapitulos, statusLeitura, limitar } from '../utils/progresso';
import { cores } from '../theme';

export default function DetalheObraScreen({ navigation, route }) {
  const { id } = route.params;
  const [obra, setObra] = useState(null);
  const [prog, setProg] = useState({ paginas_lidas: 0, capitulo_atual: 0, volume_atual: 0 });
  const [salvando, setSalvando] = useState(false);

  const carregar = useCallback(async () => {
    try {
      const o = await buscarObra(id);
      setObra(o);
      setProg({ paginas_lidas: o.paginas_lidas, capitulo_atual: o.capitulo_atual, volume_atual: o.volume_atual });
      navigation.setOptions({ title: o.titulo });
    } catch (e) {
      avisar('Erro', e.message);
    }
  }, [id]);

  useFocusEffect(useCallback(() => { carregar(); }, [carregar]));

  if (!obra) {
    return <View style={styles.centro}><ActivityIndicator size="large" color={cores.primaria} /></View>;
  }

  // Percentual recalculado em tempo real enquanto o usuário altera o progresso
  const percentual = calcularPercentual(prog.paginas_lidas, obra.total_paginas);
  const alterado =
    prog.paginas_lidas !== obra.paginas_lidas ||
    prog.capitulo_atual !== obra.capitulo_atual ||
    prog.volume_atual !== obra.volume_atual;

  const maximos = { paginas_lidas: obra.total_paginas, capitulo_atual: obra.total_capitulos, volume_atual: obra.total_volumes };
  const ajustar = (campo, delta) =>
    setProg((p) => ({ ...p, [campo]: limitar(p[campo] + delta, maximos[campo]) }));
  const digitar = (campo) => (txt) => setProg((p) => ({ ...p, [campo]: limitar(txt, maximos[campo]) }));

  async function salvar(novo = prog) {
    setSalvando(true);
    try {
      const o = await atualizarObra(id, novo);
      setObra(o);
      setProg({ paginas_lidas: o.paginas_lidas, capitulo_atual: o.capitulo_atual, volume_atual: o.volume_atual });
    } catch (e) {
      avisar('Erro ao salvar', e.message);
    } finally {
      setSalvando(false);
    }
  }

  function marcarConcluido() {
    salvar({ paginas_lidas: obra.total_paginas, capitulo_atual: obra.total_capitulos, volume_atual: obra.total_volumes });
  }

  function confirmarExclusao() {
    confirmar('Excluir obra', `Deseja excluir "${obra.titulo}"?`, 'Excluir', async () => {
      try {
        await excluirObra(id);
        navigation.goBack();
      } catch (e) {
        avisar('Erro', e.message);
      }
    });
  }

  return (
    <Fundo>
    <ScrollView contentContainerStyle={styles.container}>
      <View style={styles.cartao}>
        <Text style={styles.tipo}>{obra.tipo === 'manga' ? 'MANGÁ' : 'LIVRO'}</Text>
        <Text style={styles.titulo}>{obra.titulo}</Text>
        {!!obra.autor && <Text style={styles.autor}>{obra.autor}</Text>}

        <Text style={[styles.percentual, percentual >= 100 && { color: cores.estrela }]}>
          {percentual >= 100 ? '★ ' : ''}{percentual.toFixed(1)}%
        </Text>
        <Text style={styles.status}>{statusLeitura(percentual)}</Text>
        <BarraProgresso percentual={percentual} altura={14} mostrarTexto={false} />
        <Text style={styles.sub}>
          {prog.paginas_lidas} de {obra.total_paginas} páginas · faltam {obra.total_paginas - prog.paginas_lidas}
        </Text>
      </View>

      <View style={styles.cartao}>
        <Text style={styles.secao}>Marcar progresso</Text>
        <Contador
          label="Páginas lidas"
          valor={prog.paginas_lidas}
          max={obra.total_paginas}
          onMenos={() => ajustar('paginas_lidas', -10)}
          onMais={() => ajustar('paginas_lidas', 10)}
          passo={10}
          onDigitar={digitar('paginas_lidas')}
        />
        {obra.total_capitulos > 0 && (
          <Contador
            label="Capítulo atual"
            valor={prog.capitulo_atual}
            max={obra.total_capitulos}
            onMenos={() => ajustar('capitulo_atual', -1)}
            onMais={() => ajustar('capitulo_atual', 1)}
            onDigitar={digitar('capitulo_atual')}
            extra={`${percentualCapitulos(prog.capitulo_atual, obra.total_capitulos).toFixed(1)}% dos capítulos`}
          />
        )}
        {obra.total_volumes > 0 && (
          <Contador
            label="Volume atual"
            valor={prog.volume_atual}
            max={obra.total_volumes}
            onMenos={() => ajustar('volume_atual', -1)}
            onMais={() => ajustar('volume_atual', 1)}
            onDigitar={digitar('volume_atual')}
          />
        )}

        <TouchableOpacity
          style={[styles.botao, !alterado && styles.botaoDesativado]}
          disabled={!alterado || salvando}
          onPress={() => salvar()}
        >
          {salvando ? <ActivityIndicator color="#fff" /> : <Text style={styles.botaoTexto}>Salvar progresso</Text>}
        </TouchableOpacity>
        {percentual < 100 && (
          <TouchableOpacity style={[styles.botao, { backgroundColor: cores.estrela }]} onPress={marcarConcluido}>
            <Text style={[styles.botaoTexto, { color: cores.fundo }]}>★ Marcar como concluído</Text>
          </TouchableOpacity>
        )}
      </View>

      <View style={styles.acoes}>
        <TouchableOpacity style={styles.acao} onPress={() => navigation.navigate('Form', { id })}>
          <Text style={{ color: cores.destaque, fontWeight: '600' }}>Editar dados</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.acao} onPress={confirmarExclusao}>
          <Text style={{ color: cores.perigo, fontWeight: '600' }}>Excluir</Text>
        </TouchableOpacity>
      </View>
    </ScrollView>
    </Fundo>
  );
}

function Contador({ label, valor, max, onMenos, onMais, onDigitar, passo = 1, extra }) {
  return (
    <View style={styles.contador}>
      <Text style={styles.label}>{label} <Text style={styles.max}>(de {max})</Text></Text>
      <View style={styles.contadorLinha}>
        <TouchableOpacity style={styles.passo} onPress={onMenos}><Text style={styles.passoTexto}>−{passo > 1 ? passo : ''}</Text></TouchableOpacity>
        <TextInput style={styles.input} keyboardType="number-pad" value={String(valor)} onChangeText={onDigitar} />
        <TouchableOpacity style={styles.passo} onPress={onMais}><Text style={styles.passoTexto}>+{passo > 1 ? passo : ''}</Text></TouchableOpacity>
      </View>
      {!!extra && <Text style={styles.max}>{extra}</Text>}
    </View>
  );
}

import { ScrollView, View, Text, TextInput, TouchableOpacity, StyleSheet, ActivityIndicator } from 'react-native';
import { avisar } from '../utils/alerta';
import Fundo from '../components/Fundo';

const vazio = { titulo: '', autor: '', tipo: 'livro', total_volumes: '', total_capitulos: '', total_paginas: '' };

export default function FormObraScreen({ navigation, route }) {
  const id = route.params?.id;
  const [form, setForm] = useState(vazio);
  const [salvando, setSalvando] = useState(false);
  const [carregando, setCarregando] = useState(!!id);

  useEffect(() => {
    navigation.setOptions({ title: id ? 'Editar obra' : 'Nova obra' });
    if (!id) return;
    buscarObra(id)
      .then((o) =>
        setForm({
          titulo: o.titulo,
          autor: o.autor ?? '',
          tipo: o.tipo,
          total_volumes: String(o.total_volumes),
          total_capitulos: String(o.total_capitulos),
          total_paginas: String(o.total_paginas),
          paginas_lidas: o.paginas_lidas,
        })
      )
      .catch((e) => avisar('Erro', e.message))
      .finally(() => setCarregando(false));
  }, [id]);

  const set = (campo) => (valor) => setForm((f) => ({ ...f, [campo]: valor }));
  const num = (v) => parseInt(v, 10) || 0;

  async function salvar() {
    const totalPaginas = num(form.total_paginas);
    if (!form.titulo.trim()) return avisar('Atenção', 'Informe o título.');
    if (totalPaginas <= 0) return avisar('Atenção', 'O total de páginas deve ser maior que zero.');
    if (id && form.paginas_lidas > totalPaginas) {
      return avisar('Atenção', `Você já leu ${form.paginas_lidas} páginas; o total não pode ser menor.`);
    }

    const dados = {
      titulo: form.titulo.trim(),
      autor: form.autor.trim() || null,
      tipo: form.tipo,
      total_volumes: num(form.total_volumes),
      total_capitulos: num(form.total_capitulos),
      total_paginas: totalPaginas,
    };

    setSalvando(true);
    try {
      if (id) await atualizarObra(id, dados);
      else await criarObra(dados);
      navigation.goBack();
    } catch (e) {
      avisar('Erro ao salvar', e.message);
    } finally {
      setSalvando(false);
    }
  }

  if (carregando) {
    return <View style={styles.centro}><ActivityIndicator size="large" color={cores.primaria} /></View>;
  }

  return (
    <Fundo>
    <ScrollView contentContainerStyle={styles.container} keyboardShouldPersistTaps="handled">
      <Text style={styles.label}>Tipo</Text>
      <View style={styles.tipos}>
        {[['livro', 'Livro'], ['manga', 'Mangá']].map(([valor, rotulo]) => (
          <TouchableOpacity
            key={valor}
            style={[styles.tipo, form.tipo === valor && styles.tipoAtivo]}
            onPress={() => set('tipo')(valor)}
          >
            <Text style={[styles.tipoTexto, form.tipo === valor && { color: '#fff' }]}>{rotulo}</Text>
          </TouchableOpacity>
        ))}
      </View>

      <Campo label="Título *" value={form.titulo} onChangeText={set('titulo')} placeholder="Ex.: One Piece" />
      <Campo label="Autor" value={form.autor} onChangeText={set('autor')} placeholder="Ex.: Eiichiro Oda" />
      <View style={styles.linha}>
        <Campo flex label="Volumes" value={form.total_volumes} onChangeText={set('total_volumes')} numerico />
        <Campo flex label="Capítulos" value={form.total_capitulos} onChangeText={set('total_capitulos')} numerico />
      </View>
      <Campo label="Total de páginas *" value={form.total_paginas} onChangeText={set('total_paginas')} numerico />

      <TouchableOpacity style={styles.botao} onPress={salvar} disabled={salvando}>
        {salvando ? <ActivityIndicator color="#fff" /> : <Text style={styles.botaoTexto}>Salvar</Text>}
      </TouchableOpacity>
    </ScrollView>
    </Fundo>
  );
}

import { supabase } from '../lib/supabase';

const TABELA = 'obras';

function check({ data, error }) {
  if (error) throw new Error(error.message);
  return data;
}

export async function listarObras() {
  return check(await supabase.from(TABELA).select('*').order('atualizado_em', { ascending: false }));
}

export async function buscarObra(id) {
  return check(await supabase.from(TABELA).select('*').eq('id', id).single());
}

export async function criarObra(obra) {
  return check(await supabase.from(TABELA).insert(obra).select().single());
}

export async function atualizarObra(id, campos) {
  return check(await supabase.from(TABELA).update(campos).eq('id', id).select().single());
}

export async function excluirObra(id) {
  return check(await supabase.from(TABELA).delete().eq('id', id));
}

import { Alert, Platform } from 'react-native';

// Alert.alert não funciona no navegador; na web usamos window.alert / window.confirm.
export function avisar(titulo, mensagem) {
  if (Platform.OS === 'web') window.alert(mensagem ? `${titulo}\n\n${mensagem}` : titulo);
  else Alert.alert(titulo, mensagem);
}

export function confirmar(titulo, mensagem, textoConfirmar, aoConfirmar) {
  if (Platform.OS === 'web') {
    if (window.confirm(`${titulo}\n\n${mensagem}`)) aoConfirmar();
    return;
  }
  Alert.alert(titulo, mensagem, [
    { text: 'Cancelar', style: 'cancel' },
    { text: textoConfirmar, style: 'destructive', onPress: aoConfirmar },
  ]);
}

// Cálculo do percentual concluído (mesma regra usada na coluna gerada do banco).
export function calcularPercentual(paginasLidas, totalPaginas) {
  const lidas = Number(paginasLidas) || 0;
  const total = Number(totalPaginas) || 0;
  if (total <= 0) return 0;
  return Math.min(100, Math.round((lidas * 1000) / total) / 10);
}

export function percentualCapitulos(capituloAtual, totalCapitulos) {
  return calcularPercentual(capituloAtual, totalCapitulos);
}

export function statusLeitura(percentual) {
  if (percentual >= 100) return 'Concluído';
  if (percentual > 0) return 'Lendo';
  return 'Não iniciado';
}

// Limita um valor entre 0 e o máximo informado
export function limitar(valor, max) {
  const n = parseInt(valor, 10);
  if (Number.isNaN(n) || n < 0) return 0;
  return max > 0 ? Math.min(n, max) : n;
}
